TOTALNSA <- read.csv("TOTALNSA.csv")

TOTALNSA$observation_date <- as.Date(

# Question 1
# Mean
## The mean model forecasts future vehicle sales using the average of all historical observations. This provides a baseline forecast but does not account for changes in trend or seasonality.
mean_forecast <- meanf(TOTALNSA_ts,12)
plot(mean_forecast)

# Naive
## The naive model uses the most recent vehicle sales value to forecast all future months. It assumes sales will remain at the latest observed level.
naive_forecast <- naive(TOTALNSA_ts,12)
plot(naive_forecast)

# Random Walk With Drift
## The random walk with drift model uses the latest observation and the average historical change to forecast future vehicle sales. It accounts for an overall upward or downward trend.
rwf_forecast <- rwf(TOTALNSA_ts,12, drift = TRUE)
plot(rwf_forecast)

# Seasonal Naive
## The seasonal naive model forecasts future vehicle sales using the values from the same months in the previous year. This is useful for TOTALNSA because vehicle sales may follow recurring seasonal patterns.
snaive_forecast <- snaive(TOTALNSA_ts,12)
plot(snaive_forecast)

# Moving Average
## The five-month and nine-month moving averages smooth fluctuations in vehicle sales. The five-month average responds more quickly to changes, while the nine-month average produces a smoother trend.
MA5_forecast <- ma(TOTALNSA_ts, order = 5)
plot(MA5_forecast)

MA9_forecast <- ma(TOTALNSA_ts, order = 9)
plot(MA9_forecast)

# ETS
## The ETS model automatically selects an exponential smoothing method based on the historical data. Its forecast uses the selected level, trend, and seasonal components to predict future vehicle sales.
ets_forecast <- forecast(ets(TOTALNSA_ts),12)
plot(ets_forecast)

# Holt Winters
## The Holt-Winters model uses exponential smoothing to estimate the level, trend, and seasonal patterns in vehicle sales. Its forecast extends these estimated patterns into the next 12 months.
HW_forecast <- forecast(HoltWinters(TOTALNSA_ts),12)
plot(HW_forecast)
